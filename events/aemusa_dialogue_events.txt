# 本文件事件命名空间；所有事件 ID 均使用此前缀。
namespace = aemusa_dialogue

# 白夜馆主入口：由“联络白夜馆”法令打开，作为后续神器使领袖线的统一菜单。
country_event = {
    id = aemusa_dialogue.1 # 白夜馆主菜单事件 ID。
    title = aemusa_dialogue.1.title # 标题本地化键。
    desc = aemusa_dialogue.1.desc # 正文本地化键。
    is_triggered_only = yes # 只能由法令或其他白夜馆事件主动调用。
    picture = GFX_evt_aemusa_white_night_pavilion # 使用自定义白夜馆事件图作为主菜单背景。

    option = {
        name = aemusa_dialogue.1.aemusa # 转入爱缪莎职责通讯。
        trigger = {
            exists = event_target:aemusa_leader # 仅在爱缪莎已被召唤且仍存在时显示。
        }
        hidden_effect = {
            country_event = {
                id = aemusa_dialogue.10 # 打开既有爱缪莎职业切换事件。
                days = 1 # 延后一日打开，避免白夜馆普通事件与领袖故事窗口在同一 UI 栈中连跳。
            }
        }
    }

    option = {
        name = aemusa_dialogue.1.no_aemusa # 尚未召唤爱缪莎时的提示选项。
        trigger = {
            NOT = {
                exists = event_target:aemusa_leader # 没有可联络的爱缪莎时显示。
            }
        }
    }

    option = {
        name = aemusa_dialogue.1.roster # 查看白夜馆待接入名册。
        hidden_effect = {
            country_event = {
                id = aemusa_dialogue.2 # 打开名册预览页。
            }
        }
    }

    option = {
        name = aemusa_dialogue.1.goodbye # 关闭白夜馆通讯。
        default_hide_option = yes # 按 Esc 关闭窗口时默认选择本项。
    }
}

# 通讯对话：由“联络爱缪莎”法令主动触发。
country_event = {
    id = aemusa_dialogue.10 # 供法令引用的事件 ID。
    title = aemusa_dialogue.10.title # 标题本地化键。
    desc = aemusa_dialogue.10.desc # 正文本地化键。
    is_triggered_only = yes # 禁止随机触发，只能由法令调用。
    picture = GFX_evt_aemusa_white_night_pavilion # 提供兜底事件图，避免领袖故事窗口关闭时出现空 sprite。
    event_window_type = leader_story # 使用原版领袖故事窗口样式。

    immediate = {
        # 每次通讯均刷新肖像，让旧存档中的爱缪莎也切换为上半身版本。
        event_target:aemusa_leader = {
            aemusa_refresh_portrait = yes
        }
    }

    picture_event_data = {
        portrait = event_target:aemusa_leader # 显示已保存的爱缪莎肖像。
        room = ethic_spaceship_room # 使用原版太空船通讯背景。
    }

    option = {
        name = aemusa_dialogue.10.commander # 指挥官选项文本。
        hidden_effect = {
            aemusa_become_commander = yes # 调用指挥官切换效果。
        }
    }

    option = {
        name = aemusa_dialogue.10.official # 行政官选项文本。
        hidden_effect = {
            aemusa_become_official = yes # 调用行政官切换效果。
        }
    }

    option = {
        name = aemusa_dialogue.10.scientist # 科学家选项文本。
        hidden_effect = {
            aemusa_become_scientist = yes # 调用科学家切换效果。
        }
    }

    option = {
        name = aemusa_dialogue.10.goodbye # 不改变职业，关闭通讯。
        default_hide_option = yes # 按 Esc 关闭窗口时默认选择本项。
    }
}

# 领袖升级检测：由 on_leader_level_up 调用；此时 this 为国家，from 为刚升级的领袖。
country_event = {
    id = aemusa_dialogue.20 # 30 级肖像升级事件。
    hide_window = yes # 自动执行，不弹出额外事件窗口。
    is_triggered_only = yes # 仅由 on_action 调用。

    trigger = {
        from = {
            has_leader_flag = aemusa_unique_leader # 确保仅处理爱缪莎。
            has_base_skill >= 30 # 达到设定满级。
            NOT = {
                has_leader_flag = aemusa_level_30_portrait # 防止重复切换肖像。
            }
        }
    }

    immediate = {
        from = {
            set_leader_flag = aemusa_level_30_portrait # 记录已解锁 30 级外观。
            aemusa_refresh_portrait = yes # 立即换成 30 级专属肖像。
        }
    }
}
