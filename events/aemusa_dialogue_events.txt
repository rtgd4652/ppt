# 本文件事件命名空间；所有事件 ID 均使用此前缀。
namespace = aemusa_dialogue

# 通讯对话：由“联络爱缪莎”法令主动触发。

    id = aemusa_dialogue.10 # 供法令引用的事件 ID。
    title = aemusa_dialogue.10.title # 标题本地化键。
    desc = aemusa_dialogue.10.desc # 正文本地化键。
    is_triggered_only = yes # 禁止随机触发，只能由法令调用。
    event_window_type = leader_story # 使用原版领袖故事窗口样式。

    picture_event_data = {

    }

    option = {
        name = aemusa_dialogue.10.commander # 指挥官选项文本。
        hidden_effect = {

    }

    option = {
        name = aemusa_dialogue.10.official # 行政官选项文本。
        hidden_effect = {
            aemusa_become_official = yes # 调用行政官切换效果。
        }
    }

    option = {
        name = aemusa_dialogue.10.scientist # 科学家选项文本。
        hidden_effect = {
            aemusa_become_scientist = yes # 调用科学家切换效果。
        }
    }

    option = {
        name = aemusa_dialogue.10.goodbye # 不改变职业，关闭通讯。
        default_hide_option = yes # 按 Esc 关闭窗口时默认选择本项。
    }
}

# 领袖升级检测：由 on_leader_level_up 调用；此时 this 为国家，from 为刚升级的领袖。
country_event = {
    id = aemusa_dialogue.20 # 30 级肖像升级事件。
    hide_window = yes # 自动执行，不弹出额外事件窗口。
    is_triggered_only = yes # 仅由 on_action 调用。

    trigger = {
        from = {
            has_leader_flag = aemusa_unique_leader # 确保仅处理爱缪莎。
            has_base_skill >= 30 # 达到设定满级。
            NOT = {
                has_leader_flag = aemusa_level_30_portrait # 防止重复切换肖像。
            }
        }
    }

    immediate = {
        from = {
            set_leader_flag = aemusa_level_30_portrait # 记录已解锁 30 级外观。
            aemusa_refresh_portrait = yes # 立即换成 30 级专属肖像。
        }
    }
}
