# 指挥官专属特质：适用于爱缪莎指挥舰队时。
leader_trait_aemusa_commander = {
    cost = 0 # 专属特质不占用特质点。
    initial = no # 不在常规领袖生成界面中出现。
    randomized = no # 不允许其他领袖随机获得。
    leader_class = { commander } # 仅可由指挥官持有。

    modifier = {
        ship_weapon_damage_mult = 0.30 # 所在舰队武器伤害 +30%。
        ship_fire_rate_mult = 0.20 # 所在舰队武器射速 +20%。
    }
}
