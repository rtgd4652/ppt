# 监听所有领袖升级；对应事件会自行过滤，只处理带有爱缪莎标记的领袖。
# 作用域：this 为国家，from 为刚完成升级的领袖。
on_leader_level_up = {
    events = {
        aemusa_dialogue.20 # 达到 30 级时解锁并换用成长版立绘。
    }
}

# 监听帝国开局；事件会自行过滤，只处理“命运共鸣”起源。
on_game_start_country = {
	events = {
		aemusa_origin.1 # 为神器使起源的首都放置专属建筑和区划。
		aemusa_ms.1 # 为玩家控制的神器使国家初始化爱缪莎 1.0 主线路线账本。
		central_court.3 # 新政体登记开局元首；旧档通过白夜馆主动迁移。
	}
}

# 记录玩家完成的星系调查；根作用域为实际取得调查状态的国家。
on_system_survey = {
	events = {
		aemusa_ms.110 # 只写入“真实银河参与”事实，不自动开始第一章。
	}
}

# 记录玩家实际启动的殖民行动；根作用域为被殖民行星，由事件读取其拥有者。
on_colonization_started = {
	events = {
		aemusa_ms.111 # 与调查适配器共同构成第一章的多来源资格。
	}
}

# 命运之主被摧毁时触发重构事件；this 为舰船拥有者国家，fromfrom 为被摧毁舰船。

	events = {

	}
}
