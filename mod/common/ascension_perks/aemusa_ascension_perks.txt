# 神器共鸣之路：神器使物种线的一段飞升，负责解锁中期独立军备与经济扩展。
ap_artifact_user_resonance_path = {
	possible = {
		owner_main_species = {
			is_species_class = "ARTIFACT_USER"
		}
		custom_tooltip = {
			fail_text = "requires_ascension_perks_2"
			num_ascension_perks > 1
		}
	}

	potential = {
		owner_main_species = {
			is_species_class = "ARTIFACT_USER"
		}
		NOT = { has_ascension_perk = ap_artifact_user_resonance_path }
	}

	modifier = {
		# 当前版本不接受分学科科研速度修正，一段飞升只保留产业和舰容加成。
		country_alloys_produces_mult = 0.05
		country_consumer_goods_produces_mult = 0.05
		country_naval_cap_mult = 0.10
	}

	ai_weight = {
		factor = 15
	}
}

# 神器终解之路：神器使物种线的二段飞升，负责解锁终局超模军备。
ap_artifact_user_apotheosis_path = {
	possible = {
		owner_main_species = {
			is_species_class = "ARTIFACT_USER"
		}
		has_ascension_perk = ap_artifact_user_resonance_path
		custom_tooltip = {
			fail_text = "requires_ascension_perks_4"
			num_ascension_perks > 3
		}
	}

	potential = {
		owner_main_species = {
			is_species_class = "ARTIFACT_USER"
		}
		has_ascension_perk = ap_artifact_user_resonance_path
		NOT = { has_ascension_perk = ap_artifact_user_apotheosis_path }
	}

	modifier = {
		# 当前版本不接受分学科科研速度修正，二段飞升保留产业、凝聚力和舰队强化。
		country_alloys_produces_mult = 0.10
		country_consumer_goods_produces_mult = 0.10
		country_unity_produces_mult = 0.10
		country_naval_cap_mult = 0.20
		ship_fire_rate_mult = 0.15 # 使用已验证的射速修正，避免通用武器伤害修正报错。
	}

	ai_weight = {
		factor = 10
	}
}

# 爱缪莎命运主宰：爱缪莎个人科技线的入口，由爱缪莎达到 30 级后解锁。
ap_artifact_court_destiny = {
	possible = {
		owner_main_species = {
			is_species_class = "ARTIFACT_USER"
		}
		has_tradition = tr_artifact_court_finish
		custom_tooltip = {
			fail_text = "requires_ascension_perks_3"
			num_ascension_perks > 2
		}
		custom_tooltip = {
			fail_text = "requires_aemusa_level_30"
			exists = event_target:aemusa_leader
			event_target:aemusa_leader = {
				has_leader_flag = aemusa_unique_leader
				has_base_skill >= 30
			}
		}
	}

	potential = {
		owner_main_species = {
			is_species_class = "ARTIFACT_USER"
		}
		NOT = { has_ascension_perk = ap_artifact_court_destiny }
	}

	modifier = {
		country_edict_fund_add = 250
		# 当前版本不接受分学科科研速度修正，命运主宰飞升保留行政与舰队效果。
		country_naval_cap_mult = 0.20
		ship_home_territory_fire_rate_mult = 0.25
		ship_armor_hardening_add = 0.25
		country_leader_pool_size = 3
	}

	ai_weight = {
		factor = 20
	}
}
