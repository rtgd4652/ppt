defined_text = {
	name = GetArtifactPostcrisisLedger
	random = no
	text = {
		trigger = {
			is_variable_set = aemusa_ms_country_projections_defeated
			is_variable_set = aemusa_ms_country_anchors_archived
			is_variable_set = aemusa_ms_country_anchors_liberated
			is_variable_set = aemusa_ms_country_anchor_losses
		}
		localization_key = aph.handover.ledger
	}
	# 较早的完整结算可能没有明细，保留胜利，不用零填齐缺页。
	default = aph.handover.ledger_summary
}
