# 国家只读来源：区分危机发生事实与取得现场基准，不把它写成反制/胜利已通过。
artifact_observation_review_has_crisis_context = {
	OR = {
		has_country_flag = aemusa_ms_country_history_vanilla_crisis_started
		has_country_flag = aemusa_ms_country_anchor_baseline_recorded
	}
}

# 锚点损失计数由既有实际资源损失接口写入；必须先检查变量存在。
artifact_observation_review_has_loss_record = {
	has_country_flag = aemusa_ms_crisis_exclusive_initialized
	is_variable_set = aemusa_ms_country_anchor_losses
	check_variable = { which = aemusa_ms_country_anchor_losses value >= 1 }
}

artifact_observation_review_has_new_records = {
	has_modifier = amr_model_reviewed
	OR = {
		AND = {
			artifact_observation_review_has_crisis_context = yes
			NOT = { has_modifier = amr_crisis_context_reviewed }
		}
		AND = {
			artifact_observation_review_has_loss_record = yes
			NOT = { has_modifier = amr_loss_reviewed }
		}
	}
}

artifact_observation_review_can_append = {
	artifact_observation_review_site = yes
	owner = { artifact_observation_review_has_new_records = yes }
}
