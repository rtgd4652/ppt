# 检查单个事件簇的四阶段事实与独立结算保护是否一致。
# 调用者必须传入扩展登记表中的五个显式 key，禁止在运行时拼接名称。
aemusa_ms_cluster_state_is_consistent = {
	NOR = {
		# 进入进行中阶段前必须已经记录开始事实。
		AND = {
			has_country_flag = $IN_PROGRESS_FLAG$
			NOT = { has_country_flag = $STARTED_FLAG$ }
		}
		# 完成后不得仍处于进行中阶段，并且必须保留开始事实。
		AND = {
			has_country_flag = $COMPLETED_FLAG$
			OR = {
				has_country_flag = $IN_PROGRESS_FLAG$
				NOT = { has_country_flag = $STARTED_FLAG$ }
			}
		}
		# 结算事实必须同时拥有完成事实和独立结算保护。
		AND = {
			has_country_flag = $SETTLED_FLAG$
			OR = {
				NOT = { has_country_flag = $COMPLETED_FLAG$ }
				NOT = { has_country_flag = $SETTLEMENT_FLAG$ }
			}
		}
		# 结算保护存在时必须已经完成，避免保护标记被提前占用。
		AND = {
			has_country_flag = $SETTLEMENT_FLAG$
			NOT = { has_country_flag = $COMPLETED_FLAG$ }
		}
	}
}

	NOR = {
		AND = {
			has_country_flag = aemusa_ms_country_pause_active
			OR = {
				NOT = { is_variable_set = aemusa_ms_country_pause_reason_index }
				check_variable = { which = aemusa_ms_country_pause_reason_index value = 0 }
			}
		}
		AND = {
			NOT = { has_country_flag = aemusa_ms_country_pause_active }
			is_variable_set = aemusa_ms_country_pause_reason_index
			check_variable = { which = aemusa_ms_country_pause_reason_index value > 0 }
		}
		AND = {
			is_variable_set = aemusa_ms_country_pause_reason_index
			NOT = {
				OR = {
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 0 }
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 1 }
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 2 }
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 3 }
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 4 }
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 5 }
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 6 }
					check_variable = { which = aemusa_ms_country_pause_reason_index value = 7 }
				}
			}
		}
		AND = {
			has_country_flag = aemusa_ms_country_consistency_conflict_active
			OR = {
				NOT = { is_variable_set = aemusa_ms_country_consistency_conflict_index }
				check_variable = { which = aemusa_ms_country_consistency_conflict_index value = 0 }
			}
		}
		AND = {
			NOT = { has_country_flag = aemusa_ms_country_consistency_conflict_active }
			is_variable_set = aemusa_ms_country_consistency_conflict_index
			check_variable = { which = aemusa_ms_country_consistency_conflict_index value > 0 }
		}
	}
}
