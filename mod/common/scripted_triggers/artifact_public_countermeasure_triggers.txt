# 舰船只读筛选；已命中的剩余90日不叠加/刷新，后来者不补算。

	is_ship_class = shipclass_military
	is_disabled = no
	NOT = { has_modifier = acc_projection_disrupted }
}

# 只认原专属危机已生成的投影舰队，普通敌军和别的国家即便同名也不会被命中。
artifact_public_countermeasure_target_fleet = {
	has_fleet_flag = aemusa_ms_anchor_projection
	exists = owner
	owner = { is_country_type = aemusa_convergence }
	exists = controller
	is_civilian = no
	is_cloaked = no
	is_disabled = no
	any_owned_ship = { artifact_public_countermeasure_target_ship = yes }
}

# 调用约定为发射舰队 -> 星系 -> 目标舰队；不依赖界面/事件ROOT。
# 目标控制国 -> 发射舰队 -> 发射控制国 -> 目标控制国，最后PREV是发射控制国。
artifact_public_countermeasure_target_is_hostile = {
	controller = {
		prevprevprev = {
			controller = {
				prevprev = { is_hostile = prev }
			}
		}
	}
}

artifact_public_countermeasure_has_targets = {
	exists = solar_system
	solar_system = {
		any_fleet_in_system = {
			artifact_public_countermeasure_target_fleet = yes
			artifact_public_countermeasure_target_is_hostile = yes
		}
	}
}
