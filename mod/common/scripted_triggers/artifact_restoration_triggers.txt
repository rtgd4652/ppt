    exists = event_target:ar_an_target@root

        AND = {
            has_country_flag = ar_an_planet

        }
        AND = {
            has_country_flag = ar_an_fleet
            event_target:ar_an_target@root = { is_owned_by = root is_mobile = yes has_fleet_flag = ar_an_target@root }
        }
    }
}
