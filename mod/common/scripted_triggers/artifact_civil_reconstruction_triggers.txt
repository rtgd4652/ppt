artifact_civil_reconstruction_site = {
	artifact_civil_supply_safe = yes
	has_active_building = "building_artifact_post_disaster_reconstruction_workshop"
	owner = { has_technology = "tech_artifact_civilization_joint_recovery" }
}

artifact_civil_reconstruction_project = {
	has_modifier = acr_rebuild_project
	artifact_civil_supply_original_owner = yes
}

artifact_civil_reconstruction_active = {
	artifact_civil_reconstruction_project = yes
	artifact_civil_reconstruction_site = yes
}

artifact_civil_reconstruction_can_start = {
	artifact_civil_reconstruction_site = yes
	artifact_civil_supply_stock = yes
	# 读取真实本地荒废度，不用战争/危机旗标推断每个殖民地受损。
	planet_devastation >= 10
	owner = {
		has_resource = { type = energy amount >= 500 }
		has_resource = { type = consumer_goods amount >= 200 }
		# 暂停项目仍占全国名额；已收尾的地方记录不占位。
		NOT = { any_owned_colony = { carrier = { has_modifier = acr_rebuild_project } } }
	}
}

# 失控或工场停用后仍可放弃，已支付材料及已完成恢复都不回滚。
artifact_civil_reconstruction_can_cancel = {
	artifact_civil_supply_actor = yes
	artifact_civil_reconstruction_project = yes
}
