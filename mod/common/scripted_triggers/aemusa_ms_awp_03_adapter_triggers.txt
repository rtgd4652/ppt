# AWP-03 只读来源与战争资格；原版关系修正和战争作用域是唯一事实依据。
# 候选国家必须在原版关系上留下对当前玩家的要求回应。
aemusa_ms_fe_has_demand_response_with_player = {
	is_country_type = fallen_empire
	has_communications = root
	OR = {
		has_opinion_modifier = { who = root modifier = opinion_refused_fallen_empire_demand }
		has_opinion_modifier = { who = root modifier = opinion_refused_fallen_empire_request }
		has_opinion_modifier = { who = root modifier = opinion_accepted_fallen_empire_request }
	}
}

# 同一主体可能先后留下多种回应修正；不得把其中一种写成唯一来源。
aemusa_ms_fe_has_multiple_demand_response_types_with_player = {
	OR = {
		AND = {
			has_opinion_modifier = { who = root modifier = opinion_refused_fallen_empire_demand }
			has_opinion_modifier = { who = root modifier = opinion_refused_fallen_empire_request }
		}
		AND = {
			has_opinion_modifier = { who = root modifier = opinion_refused_fallen_empire_demand }
			has_opinion_modifier = { who = root modifier = opinion_accepted_fallen_empire_request }
		}
		AND = {
			has_opinion_modifier = { who = root modifier = opinion_refused_fallen_empire_request }
			has_opinion_modifier = { who = root modifier = opinion_accepted_fallen_empire_request }
		}
	}
}

# 只在唯一候选存在时锁定主体；多个真实来源并存时不得随机指定叙事对象。
aemusa_ms_country_has_unrecorded_fe_intervention = {
	has_country_flag = aemusa_ms_country_route_initialized
	NOT = { has_country_flag = aemusa_ms_country_fe_intervention_recorded }
	count_country = {
		count = 1
		limit = { aemusa_ms_fe_has_demand_response_with_player = yes }
	}
}

# 歧义只表示尚不能可靠选择哪一个失落帝国，关系修正不因此被抹去。
aemusa_ms_country_has_ambiguous_fe_intervention = {
	has_country_flag = aemusa_ms_country_route_initialized
	NOT = { has_country_flag = aemusa_ms_country_fe_intervention_recorded }
	count_country = {
		count > 1
		limit = { aemusa_ms_fe_has_demand_response_with_player = yes }
	}
}

# 调用位置：on_war_beginning 的国家事件，root 为玩家、from 为本次战争。
# 宣战本身锁定真实攻击者；不以前置外交回应或第四章结算倒推战争资格。
aemusa_ms_country_can_record_fe_war_start = {
	is_ai = no
	has_country_flag = aemusa_ms_country_route_initialized
	NOT = { has_country_flag = aemusa_ms_country_history_fallen_empire_war_started }
	from = {
		NOT = { is_from_proxy_war = yes }
		any_attacker = {
			is_country_type = awakened_fallen_empire
			is_war_leader = yes
		}
		any_defender = {
			is_same_value = root
			is_war_leader = yes
		}
	}
}
