# R2c轨道观测接口：只作用于本座正常星港，不替换舰载传感器或创建轨道巨构。
artifact_orbital_observation_link = {
	icon = "GFX_spaceport_module_navigators_guild"
	construction_days = 360
	category = sm_utility_cat
	starbase_type = starbase
	starbase_limit = {
		base = 1
	}
	show_in_tech = "tech_artifact_orbital_coordination"
	potential = {
		is_normal_starbase = yes
		exists = owner
		owner = {
			has_technology = "tech_artifact_orbital_coordination"
			owner_main_species = {
				is_species_class = "ARTIFACT_USER"
			}
		}
	}
	possible = {
		has_starbase_size >= starbase_starport
		is_controlled_by = owner
	}
	resources = {
		category = starbase_buildings
		cost = {
			alloys = 180
		}
		upkeep = {
			energy = 4
			alloys = 1
		}
	}
	# 星港→拥有国→回到星港→星系→殖民地；最后prevprevprev明确指向拥有国。
	# 只取同星、本国、未被占领的有效阵列，不能借另一帝国或别的星系满足。
	triggered_station_modifier = {
		potential = {
			exists = owner
			is_controlled_by = owner
			owner = {
				has_technology = "tech_artifact_orbital_coordination"
				has_technology = "tech_artifact_long_baseline_measurement"
				owner_main_species = {
					is_species_class = "ARTIFACT_USER"
				}
				prev = {
					solar_system = {
						any_system_planet_colony = {
							is_owned_by = prevprevprev
							is_occupied_flag = no
							OR = {
								has_active_building = "building_artifact_long_baseline_array"
								has_active_building = "building_artifact_phase_warning_array"
								# A09把付费地方观测端作为替代来源；共用本块，不能叠加一份基准支持。
								owner = {
									has_technology = "tech_artifact_cross_colony_calibration"
									has_modifier = aoc_calibration_archive
									prev = {
										carrier = { artifact_observation_calibration_window_operational = yes }
									}
								}
							}
						}
					}
				}
			}
		}
		ship_sensor_range_add = 1
		ship_hyperlane_range_add = 2
	}
	# 高级阵列或有档案的F02付费监听提供一份本星港探测；不揭示未来或绕过星云。
	triggered_station_modifier = {
		potential = {
			exists = owner
			is_controlled_by = owner
			owner = {
				has_technology = "tech_artifact_orbital_coordination"
				owner_main_species = {
					is_species_class = "ARTIFACT_USER"
				}
				prev = {
					solar_system = {
						any_system_planet_colony = {
							is_owned_by = prevprevprev
							is_occupied_flag = no
							# A07原阵列和F02付费监听是同一份探测供源，不叠加。
							OR = {
								AND = {
									owner = { has_technology = "tech_artifact_phase_disturbance_warning" }
									has_active_building = "building_artifact_phase_warning_array"
								}
								carrier = { artifact_crisis_intelligence_listening_active = yes }
							}
						}
					}
				}
			}
		}
		ship_cloaking_detection_add = 1
	}
	# D14复用既有轨道接口。本星系至少一个有效本国联防殖民地即提供一份星港收益；
	# 不按殖民地数叠加，不作用于机动舰队或防御平台，原观测收益继续独立判定。
	triggered_station_modifier = {
		potential = {
			exists = owner
			owner = { has_technology = "tech_artifact_civilization_joint_defense" }
			artifact_joint_defense_station_active = yes
		}
		ship_fire_rate_mult = 0.15
		ship_tracking_add = 10
	}
	ai_weight = {
		weight = 10
		modifier = {
			factor = 0
			NOT = {
				owner = {
					prev = {
						solar_system = {
							any_system_planet_colony = {
								is_owned_by = prevprevprev
								is_occupied_flag = no
								OR = {
									has_active_building = "building_artifact_long_baseline_array"
									has_active_building = "building_artifact_phase_warning_array"
								}
							}
						}
					}
				}
			}
		}
	}
}
