# 一次性法令：支付凝聚力后招募一名 11 级官员领袖。

edict_summon_stellar_administrator = {
    length = 0 # 0 表示即时生效、不会持续占用法令栏。
    icon = "GFX_edict_type_policy" # 临时使用原版占位图标，避免自定义图标 sprite 未注册时崩溃。

    resources = {
        category = edicts # 使用法令资源类别，确保 UI 正确显示花费。
        cost = {
            unity = 500 # 基础花费：500 凝聚力。
            multiplier = value:edict_size_effect # 花费随帝国规模修正。
        }
    }

    potential = {
        owner_main_species = {
            is_species_class = "ARTIFACT_USER" # 仅神器使主物种可见并能召唤爱缪莎。
        }
    }

    allow = {
        owner_main_species = {
            is_species_class = "ARTIFACT_USER" # 再次确认帝国主物种，防止条件在 UI 更新前变化。
        }
        # 已存在爱缪莎时禁止再次召唤，避免重复生成。
        NOT = {
            exists = event_target:aemusa_leader
        }
        # 凝聚力是否充足由上方 resources.cost 自动检查与扣除。
    }

    ai_weight = {
        weight = 0 # 不让 AI 自动使用，避免生成无主的特殊领袖。
    }

    effect = {
        hidden_effect = {
            aemusa_summon_leader = yes # 调用脚本化效果创建领袖，避免法令文件过于冗长。
        }
    }
}

	length = 0 # 即时触发，不占用法令栏；永久效果通过国家修正保存。
	icon = "GFX_edict_type_policy" # 临时使用原版占位图标，避免自定义图标 sprite 未注册时崩溃。

	potential = {
		is_ai = no # 仅玩家可见，避免 AI 测试加成污染正常局势。
	}

	allow = {
		always = yes # 测试按钮不设置花费和门槛，便于随时刷新效果。
	}

	effect = {
		add_modifier = {
			modifier = aemusa_debug_overdrive # 添加永久测试修正。
			days = -1 # -1 表示永久持续。

		add_resource = {

			rare_crystals = 10000000 # 测试稀有水晶。
			unity = 100000000 # 测试凝聚力。
			influence = 5000000 # 测试影响力。
			minor_artifacts = 5000000 # 测试小型文物。
			physics_research = 100000000 # 测试物理学点数。
			society_research = 100000000 # 测试社会学点数。
			engineering_research = 100000000 # 测试工程学点数。
		}

		every_owned_leader = {
			limit = {
				OR = {
					has_leader_flag = aemusa_unique_leader # 爱缪莎。
					has_leader_flag = seth_unique_leader # 赛斯。
					has_leader_flag = yutong_unique_leader # 幽桐。

					has_trait = leader_trait_yutong_commander # 幽桐指挥官形态。
					has_trait = leader_trait_rabi_commander # 拉比指挥官形态。
				}
			}
			aemusa_debug_raise_artifact_leader_to_30 = yes # 将所有已接入神器使领袖补到 30 级。
		}
	}

	ai_weight = {
		weight = 0 # 测试法令禁止 AI 使用。
	}
