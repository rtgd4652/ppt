# 科研分区：原生决议正常扣费，状态只保存方向，效益由当地演算设施提供。
decision_artifact_city_plan_research = {
	owned_planets_only = yes
	sound = "event_administrative_work"
	icon = "decision_artifact_destiny_city_project"
	resources = {
		category = decisions
		cost = { energy = 300 unity = 100 }
	}
	potential = {
		is_colony = yes
		exists = owner
		owner = {
			owner_main_species = { is_species_class = "ARTIFACT_USER" }
			has_technology = "tech_artifact_city_functional_zoning"
		}
		NOR = {
			has_modifier = artifact_city_plan_research
			has_modifier = artifact_city_plan_production
			has_modifier = artifact_city_plan_services
		}
	}
	allow = {
		custom_tooltip = {
			fail_text = artifact_city_plan_requires_control
			is_controlled_by = owner
		}
		custom_tooltip = {
			fail_text = artifact_city_plan_requires_active_facility
			OR = {
				has_active_building = "building_artifact_city_calculation_room"
				has_active_building = "building_artifact_fate_calculus_institute"
			}
		}
	}
	effect = {
		add_modifier = { modifier = artifact_city_plan_research days = -1 }
	}
	ai_weight = { weight = 0 }
}
