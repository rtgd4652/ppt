# 安、格蕾莎、雯梓、丽：单任务、有限结果、同实体职业。
ar_an_recruit = {
    if = {
        limit = { ar_an_recruit_available = yes has_resource = { type = unity amount >= 500 } }
        add_resource = { unity = -500 }
        create_leader = {
            class = official
            name = ar.an.name
            gender = female
            species = owner_main_species

            custom_description = ar.an.background
            event_leader = yes
            skill = 7
            randomize_traits = no
            skip_background_generation = yes
            traits = { trait = leader_trait_ar_an_identity }
            effect = { set_leader_flag = ar_an_unique save_global_event_target_as = ar_an@root }
        }
        set_country_flag = ar_an_recruited
        log = "RESTORATION|an_recruited"
    }
}

ar_an_become_official = {
    # 同实体转职，任务、经验、费用与恢复期均不写入。
    ar_rebind = yes
    if = {
        limit = { ar_player = yes ar_an_available = yes NOT = { ar_an_official = yes } }
        event_target:ar_an@root = {
            unassign_leader = yes
            remove_trait = leader_trait_ar_an_identity
            change_leader_class = official
            add_trait = { trait = leader_trait_ar_an_identity show_message = no }
        }
    }
}

ar_an_clear_task = {
    # 清理关系目标，保留已付成本、最后进度、归档次数和独立恢复期。
    if = {
        limit = { exists = event_target:ar_an_target@root }
        if = { limit = { has_country_flag = ar_an_planet } event_target:ar_an_target@root = { remove_planet_flag = ar_an_target@root } }
        if = { limit = { has_country_flag = ar_an_fleet } event_target:ar_an_target@root = { remove_fleet_flag = ar_an_target@root } }
        clear_global_event_target = ar_an_target@root
    }
    remove_country_flag = ar_an_active
    remove_country_flag = ar_an_ready
    remove_country_flag = ar_an_working
    remove_country_flag = ar_an_record_ready
    remove_country_flag = ar_an_risk
    remove_country_flag = ar_an_planet
    remove_country_flag = ar_an_fleet
}

ar_an_withdraw = {
    if = {
        limit = { OR = { has_country_flag = ar_an_active has_country_flag = ar_an_ready has_country_flag = ar_an_working has_country_flag = ar_an_record_ready } }
        ar_an_clear_task = yes
        set_country_flag = ar_an_withdrawn
        if = { limit = { NOT = { has_country_flag = ar_an_cooldown } } set_timed_country_flag = { flag = ar_an_cooldown days = 90 } }
        log = "RESTORATION|an_withdrawn_costs_retained"
    }
}

ar_an_start_planet = {
    if = {
        limit = { ar_an_planet_start_available = yes }

        if = {
            limit = { exists = capital_scope capital_scope = { is_colony = yes is_controlled_by = root } }
            capital_scope = { set_planet_flag = ar_an_target@root save_global_event_target_as = ar_an_target@root }
        }
        else = {
            random_owned_planet = { limit = { is_colony = yes is_controlled_by = root } set_planet_flag = ar_an_target@root save_global_event_target_as = ar_an_target@root }
        }

        if = {
            limit = { exists = event_target:ar_an_target@root }
            add_resource = { energy = -100 }
            add_resource = { unity = -50 }
            set_variable = { which = ar_an_cost_energy value = 100 }
            set_variable = { which = ar_an_cost_unity value = 50 }
            set_variable = { which = ar_an_cost_consumer_goods value = 0 }
            set_variable = { which = ar_an_cost_minerals value = 0 }
            set_variable = { which = ar_an_cost_alloys value = 0 }
            set_country_flag = ar_an_planet
            set_country_flag = ar_an_active
            remove_country_flag = ar_an_completed
            remove_country_flag = ar_an_withdrawn
            set_variable = { which = ar_an_progress value = 0 }
            remove_country_flag = ar_an_combat_observed
            if = { limit = { exists = event_target:ar_an_observed_enemy@root } clear_global_event_target = ar_an_observed_enemy@root }
            log = "RESTORATION|an_planet_registered"
        }
    }
}

    }
