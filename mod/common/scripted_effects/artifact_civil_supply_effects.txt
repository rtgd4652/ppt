# 沿用国家月度入口，不新增全银河扫描。停工不增加工作量，也不追补月份。
artifact_civil_supply_hull_monthly = {
	if = {
		limit = { artifact_civil_supply_hull_project = yes }
		if = {
			limit = { owner = { has_modifier = acs_hull_qualified } }
			artifact_civil_supply_hull_clear = yes
		}
		else_if = {
			limit = { artifact_civil_supply_hull_site = yes }
			change_variable = { which = acs_hull_work value = 1 }
			if = {
				limit = { check_variable = { which = acs_hull_work value >= 6 } }
				# 结果不含战斗数值，只作为新组件可装配的国家资格；不会授予其他研究。
				owner = { add_modifier = { modifier = acs_hull_qualified days = -1 } }
				artifact_civil_supply_hull_clear = yes
			}
		}
	}
}

# 原月度入口调用；if/else_if保证单次只推进一个已付阶段，等待确认不推进、不补跑。
artifact_civil_supply_segmented_monthly = {
	if = {
		limit = {
			artifact_civil_supply_segmented_project = yes
			# 应急支援只替代已付阶段的物流资格，核心营建场/本国控制仍必需。
			# 新开工、阶段付款及完成容量继续使用原条件；不加速工作量或越过验收。
			artifact_civil_supply_segmented_work_site = yes
		}
		if = {
			limit = { has_modifier = acs_segmented_preparing }
			change_variable = { which = acs_segmented_work value = 1 }
			if = {
				limit = { check_variable = { which = acs_segmented_work value >= 3 } }
				remove_modifier = acs_segmented_preparing
				add_modifier = { modifier = acs_segmented_prepared days = -1 }
				set_variable = { which = acs_segmented_work value = 0 }
			}
		}
		else_if = {
			limit = { has_modifier = acs_segmented_assembling }
			change_variable = { which = acs_segmented_work value = 1 }
			if = {
				limit = { check_variable = { which = acs_segmented_work value >= 4 } }
				remove_modifier = acs_segmented_assembling
				add_modifier = { modifier = acs_segmented_assembled days = -1 }
				set_variable = { which = acs_segmented_work value = 0 }
			}
		}
		else_if = {
			limit = { has_modifier = acs_segmented_commissioning }
			change_variable = { which = acs_segmented_work value = 1 }
			if = {
				limit = { check_variable = { which = acs_segmented_work value >= 2 } }
				remove_modifier = acs_segmented_commissioning
				add_modifier = { modifier = acs_segmented_operational days = -1 }
				clear_variable = acs_segmented_work
			}
		}
	}
}
