# 固定物资由效果实际逐项扣除；先移除库存再退款／转运，调用重复也不重复结算。
# 决议不另收原生费用，避免折扣后按固定面额退款的套利。
artifact_civil_supply_clear = {
	remove_modifier = acs_supply_reserved
	remove_modifier = acs_link_source
	remove_modifier = acs_link_receiver
	remove_modifier = acs_refinery_motes
	remove_modifier = acs_refinery_crystals
	remove_modifier = acs_refinery_gases
	remove_modifier = acs_refinery_adjusting
	remove_modifier = acs_civic_industry
	remove_modifier = acs_civic_research
	remove_modifier = acs_civic_services
	remove_modifier = acs_civic_adjusting
	remove_modifier = acs_construction_supply
	remove_carrier_flag = acs_managed
	if = { limit = { exists = event_target:acs_owner@this } clear_global_event_target = acs_owner@this }
}

artifact_civil_supply_ensure_owner = {
	if = { limit = { artifact_civil_supply_original_owner = no } artifact_civil_supply_clear = yes set_carrier_flag = acs_managed owner = { save_global_event_target_as = acs_owner@prev } }
}

artifact_civil_supply_reserve = {
	if = {
		limit = {
			artifact_civil_supply_can_reserve = yes
		}
		artifact_civil_supply_ensure_owner = yes
		owner = {
			add_resource = {
				minerals = -500
			}

			add_resource = {
				alloys = -100
			}
		}
		add_modifier = { modifier = acs_supply_reserved days = -1 }
	}
}

artifact_civil_supply_transfer = {
	if = {
		limit = {
			artifact_civil_supply_can_transfer = yes
		}
		owner = {
			add_resource = {
				energy = -100
			}
		}
		remove_modifier = acs_supply_reserved
		owner = {
			random_owned_colony = {
				limit = {
					carrier = { artifact_civil_supply_receiver = yes NOT = { has_modifier = acs_supply_reserved } }
				}
				carrier = { add_modifier = { modifier = acs_supply_reserved days = -1 } }
			}
		}
	}
}

		limit = {
			artifact_civil_supply_can_megastructure = yes
		}
		owner = {
			add_resource = {
				energy = -1000
			}

			add_resource = {
				alloys = -300
			}
		}

		owner = {
			random_owned_colony = {

				}

			}

		}
	}
}
