# 清理只涉及本包现场，不退款、不操作角色工程或C2修正。
artifact_foundation_clear = {
	remove_carrier_flag = arf_r1_active
	remove_carrier_flag = arf_r1_paid
	remove_carrier_flag = arf_r1_managed
	remove_carrier_flag = arf_r1_completed
	remove_carrier_flag = arf_r1_mode_sample_analysis
	remove_carrier_flag = arf_r1_mode_basic_construction
	remove_carrier_flag = arf_r1_mode_precision_construction
	remove_carrier_flag = arf_r1_mode_materials_study
	remove_carrier_flag = arf_r1_mode_reproduce_sample
	remove_carrier_flag = arf_r1_mode_reproduce_materials
	remove_carrier_flag = arf_r1_mode_boundary_experiment
	remove_modifier = artifact_foundation_boundary_result
	clear_variable = arf_r4_preparation
	clear_variable = arf_r4_experiment
	clear_variable = arf_r4_recovery
	remove_modifier = artifact_foundation_sample_analysis
	remove_modifier = artifact_foundation_basic_construction
	remove_modifier = artifact_foundation_precision_construction
	remove_modifier = artifact_foundation_materials_study
	clear_variable = arf_r1_progress
	clear_variable = arf_r1_required_work
	clear_global_event_target = arf_r1_owner@this
}

artifact_foundation_begin = {
	artifact_foundation_clear = yes
	remove_carrier_flag = arf_r1_failed
	set_carrier_flag = arf_r1_managed
	set_carrier_flag = arf_r1_active
	set_carrier_flag = arf_r1_paid
	set_variable = {
		which = arf_r1_progress
		value = 0
	}
	owner = {
		save_global_event_target_as = arf_r1_owner@prev
	}
}

artifact_foundation_abort = {
	# 失去现场或主动中止时保留已完成阶段，不能把失败记录作为有效成果。
	if = {
		limit = { has_carrier_flag = arf_r1_mode_boundary_experiment }
		set_variable = { which = arf_r4_last_preparation value = arf_r4_preparation }
		set_variable = { which = arf_r4_last_experiment value = arf_r4_experiment }
		set_variable = { which = arf_r4_last_recovery value = arf_r4_recovery }
	}
	if = {
		limit = {
			has_carrier_flag = arf_r1_active
		}
		set_variable = {
			which = arf_r1_last_progress
			value = arf_r1_progress
		}
		set_carrier_flag = arf_r1_failed
	}
	artifact_foundation_clear = yes
}

# 必须仍在进行且已付费，先清进行态，再只结算一个方案。
artifact_foundation_complete = {
	if = {
		limit = {
			has_carrier_flag = arf_r1_active
			has_carrier_flag = arf_r1_paid
			artifact_foundation_work_ready = yes
			check_variable = {
				which = arf_r1_progress
				value >= arf_r1_required_work
			}
		}
		remove_carrier_flag = arf_r1_active
		remove_carrier_flag = arf_r1_paid
		set_carrier_flag = arf_r1_completed
		if = {
			limit = { has_carrier_flag = arf_r1_mode_boundary_experiment }
			add_modifier = { modifier = artifact_foundation_boundary_result days = 720 }
		}
		else_if = {
			limit = {
				OR = {
					has_carrier_flag = arf_r1_mode_sample_analysis
					has_carrier_flag = arf_r1_mode_reproduce_sample
				}
			}
			add_modifier = {
				modifier = artifact_foundation_sample_analysis
				days = 720
			}
		}
		else_if = {
			limit = {
				has_carrier_flag = arf_r1_mode_basic_construction
			}
			add_modifier = {
				modifier = artifact_foundation_basic_construction
				days = 720
			}
		}
		else_if = {
			limit = {
				has_carrier_flag = arf_r1_mode_precision_construction
			}
			add_modifier = {
				modifier = artifact_foundation_precision_construction
				days = 720
			}
		}
		else_if = {
			limit = {
				OR = {
					has_carrier_flag = arf_r1_mode_materials_study
					has_carrier_flag = arf_r1_mode_reproduce_materials
				}
			}
			add_modifier = {
				modifier = artifact_foundation_materials_study
				days = 720
			}
		}
		remove_carrier_flag = arf_r1_mode_sample_analysis
		remove_carrier_flag = arf_r1_mode_basic_construction
		remove_carrier_flag = arf_r1_mode_precision_construction
		remove_carrier_flag = arf_r1_mode_materials_study
		remove_carrier_flag = arf_r1_mode_reproduce_sample
		remove_carrier_flag = arf_r1_mode_reproduce_materials
		remove_carrier_flag = arf_r1_mode_boundary_experiment
	}
}

# 只有原生月度入口调用此效果，打开界面不调用它。
artifact_foundation_monthly_planet = {
	if = {
		limit = {
			OR = {
				artifact_foundation_actor = no
				artifact_foundation_original_owner = no
			}
		}
		artifact_foundation_abort = yes
	}
	else_if = {
		limit = {
			has_carrier_flag = arf_r1_active
		}
		if = {
			limit = {
				NOT = {
					has_carrier_flag = arf_r1_paid
				}
			}
			artifact_foundation_abort = yes
		}
		else_if = {
			limit = {
				artifact_foundation_work_ready = yes
			}
			if = {
				limit = { has_carrier_flag = arf_r1_mode_boundary_experiment }
				artifact_foundation_advance_boundary = yes
			}
			change_variable = {
				which = arf_r1_progress
				value = 1
			}
			artifact_foundation_complete = yes
		}
	}
	else_if = {
		limit = {
			NOR = {
				has_modifier = artifact_foundation_sample_analysis
				has_modifier = artifact_foundation_basic_construction
				has_modifier = artifact_foundation_precision_construction
				has_modifier = artifact_foundation_materials_study
				has_modifier = artifact_foundation_boundary_result
			}
		}
		artifact_foundation_clear = yes
	}
}

# 付费、共享槽位与原拥有者建账一次完成；窗口重入只读取现场。
artifact_foundation_start_boundary = {
	if = {
		limit = { artifact_foundation_can_start_boundary = yes }
		owner = {
			add_resource = { energy = -600 }
			add_resource = { consumer_goods = -200 }
			add_resource = { alloys = -100 }
		}
		artifact_foundation_begin = yes
		set_carrier_flag = arf_r1_mode_boundary_experiment
		set_variable = { which = arf_r1_required_work value = 6 }
		set_variable = { which = arf_r4_preparation value = 0 }
		set_variable = { which = arf_r4_experiment value = 0 }
		set_variable = { which = arf_r4_recovery value = 0 }
	}
}

# 一个有效月度只推进一个阶段，避免同月从准备直接跳到结算。
artifact_foundation_advance_boundary = {
	if = {
		limit = { NOT = { check_variable = { which = arf_r4_preparation value >= 2 } } }
		change_variable = { which = arf_r4_preparation value = 1 }
	}
	else_if = {
		limit = { NOT = { check_variable = { which = arf_r4_experiment value >= 3 } } }
		change_variable = { which = arf_r4_experiment value = 1 }
	}
	else = { change_variable = { which = arf_r4_recovery value = 1 } }
}

# 公共接口归国家，消耗本地真实记录；不触碰主线枢纽或个人转变。
artifact_boundary_archive = {

		owner = { set_country_flag = artifact_public_rule_interface_record }
		artifact_foundation_clear = yes
	}
}
