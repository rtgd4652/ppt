# 创建爱缪莎。她初始为行政官，并在创建时保存为全局事件目标。
aemusa_summon_leader = {
    create_leader = {
        class = official # 初始职业：行政官。
        name = "aemusa" # 使用本地化键显示姓名“爱缪莎”。
        gender = female # 领袖性别。
        species = owner_main_species # 使用帝国主物种，避免创建领袖时出现 species: none 错误。
        custom_description = aemusa_leader_background # 使用中央庭七人众的专属来历，替代默认 job_ 文本。
        background_ethic = ethic_spiritualist # 以唯心背景强调其与神器、命运和中央庭的联系。
        event_leader = yes # 标记为事件领袖，便于作为独特角色管理。
        skill = 11 # 初始等级为 11 级；满级由全局定义调整为 30 级。
        randomize_traits = no # 禁止随机特质，确保只拥有专属技能。
        skip_background_generation = yes # 不生成随机背景故事。
        traits = {
            trait = leader_trait_aemusa_official # 初始专属技能：命运织网。
        }
        effect = {
            aemusa_refresh_portrait = yes # 根据等级选择城市通讯立绘或 30 级专属立绘。
            set_leader_flag = aemusa_unique_leader # 供后续逻辑识别此独特领袖。
            save_global_event_target_as = aemusa_leader # 供通讯法令和事件持续定位该领袖。
        }
    }
}

# 三种职责共用同一人物。重复选择当前职业不解除任命，不重建领袖或写入成长与主线。
aemusa_change_duty = {
    if = {
        limit = {
            exists = event_target:aemusa_leader
            event_target:aemusa_leader = {
                has_leader_flag = aemusa_unique_leader
                NOT = { leader_class = $CLASS$ }
            }
        }
        event_target:aemusa_leader = {
            unassign_leader = this # 先结束原岗位，转职后由玩家重新任命。
            remove_trait = leader_trait_aemusa_commander
            remove_trait = leader_trait_aemusa_official
            remove_trait = leader_trait_aemusa_scientist
            change_leader_class = $CLASS$
            add_trait = {
                trait = $TRAIT$
                show_message = no
                consume_selection = yes
            }
            aemusa_refresh_portrait = yes
        }
    }
}

# 保留既有调用名称，职业专属技能随同一人物切换。
aemusa_become_commander = {
    aemusa_change_duty = { CLASS = commander TRAIT = leader_trait_aemusa_commander }
}
aemusa_become_official = {
    aemusa_change_duty = { CLASS = official TRAIT = leader_trait_aemusa_official }
}
aemusa_become_scientist = {
    aemusa_change_duty = { CLASS = scientist TRAIT = leader_trait_aemusa_scientist }
}

# 按爱缪莎当前等级刷新肖像：30 级使用成长后的专属立绘，其余等级使用城市通讯立绘。
aemusa_refresh_portrait = {
    if = {
        limit = {
            OR = {
                has_leader_flag = aemusa_level_30_portrait
                has_base_skill >= 30 # 旧档或其他正常成长来源也可恢复已达到的外观。
            }
        }
        set_leader_flag = aemusa_level_30_portrait
        change_leader_portrait = aemusa_portrait_level_30 # 30 级专属肖像。
    }
    else = {
        change_leader_portrait = aemusa_portrait # 常规城市背景肖像。
    }
}

# 创建赛斯。赛斯作为圣星教会神官及中央庭协作者接入，初始为行政官并保存为全局事件目标。

            custom_description = seth_leader_background # 使用圣星教会神官与中央庭协作的背景文本。
            background_ethic = ethic_spiritualist # 以唯心背景强调神官、信仰与守护者身份。
            event_leader = yes # 标记为事件领袖，便于作为独特角色管理。
            skill = 7 # 初始等级为 7 级，低于爱缪莎但高于普通招募领袖。

            skip_background_generation = yes # 不生成随机背景故事。
            traits = {
                trait = leader_trait_seth_official # 初始专属技能：羽蛇神杖。

            }
        }
    }
}
