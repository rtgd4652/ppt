artifact_growth_read_aemusa = {
    set_variable = {
        which = ag_aemusa_stage
        value = 0
    }
    set_variable = {
        which = ag_aemusa_primary
        value = 0
    }
    set_variable = {
        which = ag_aemusa_support
        value = 0
    }
    if = {
        limit = {
            any_owned_leader = {
                has_leader_flag = aemusa_unique_leader
                has_base_skill >= 21
            }
        }
        set_variable = {
            which = ag_aemusa_stage
            value = 3
        }
        set_variable = {
            which = ag_aemusa_primary
            value = 1.5
        }
        set_variable = {
            which = ag_aemusa_support
            value = 1.3
        }
    }
    else_if = {
        limit = {
            any_owned_leader = {
                has_leader_flag = aemusa_unique_leader
                has_base_skill >= 11
            }
        }
        set_variable = {
            which = ag_aemusa_stage
            value = 2
        }
        set_variable = {
            which = ag_aemusa_primary
            value = 1.25
        }
        set_variable = {
            which = ag_aemusa_support
            value = 1.2
        }
    }
    else_if = {
        limit = {
            any_owned_leader = {

            }

        set_variable = {
            which = ag_aemusa_stage

            which = ag_aemusa_primary
            value = 1
        }

    }
}
