# AWP-03 只读诊断：分别报告事实、等待和攻击者目标缺失，不把尚未发生的战争算作失败。
# 控制台在玩家国家作用域执行：effect aemusa_ms_log_awp_03_audit = yes
aemusa_ms_log_awp_03_audit = {
	log = "AEMUSA-MS|AWP-03|audit begin; scope=country"
	if = {
		limit = {
			has_country_flag = aemusa_ms_country_route_initialized
			aemusa_ms_country_has_valid_fallen_empire_war_history = yes
		}
		log = "AEMUSA-MS|PASS|awp_03_registered_state"
	}
	else = { log = "AEMUSA-MS|FAIL|awp_03_registered_state" }

	if = {
		limit = {
			has_country_flag = aemusa_ms_country_history_fallen_empire_war_started
			exists = event_target:aemusa_ms_fe_war_attacker@root
		}
		log = "AEMUSA-MS|OBSERVED|awp_03_war_attacker_locked"
	}
	else_if = {
		limit = { has_country_flag = aemusa_ms_country_history_fallen_empire_war_started }
		log = "AEMUSA-MS|BLOCK|awp_03_war_attacker_missing"
	}
	else = { log = "AEMUSA-MS|WAIT|awp_03_war_not_started" }

	if = {
		limit = {
			has_country_flag = aemusa_ms_country_history_fallen_empire_war_started
			any_war = { has_war_flag = aemusa_ms_qualified_fe_war }
		}
		log = "AEMUSA-MS|OBSERVED|awp_03_qualified_war_active"
	}
	else_if = {
		limit = {
			has_country_flag = aemusa_ms_country_history_fallen_empire_war_started
			NOT = { has_country_flag = aemusa_ms_country_history_fallen_empire_war_completed }
		}
		log = "AEMUSA-MS|CHECK|awp_03_started_war_not_active"
	}
	else_if = {
		limit = { has_country_flag = aemusa_ms_country_history_fallen_empire_war_started }
		log = "AEMUSA-MS|OBSERVED|awp_03_qualified_war_ended"
	}
	else = { log = "AEMUSA-MS|WAIT|awp_03_qualified_war_not_started" }

	if = {
		limit = { has_country_flag = aemusa_ms_country_fe_war_player_victory }
		log = "AEMUSA-MS|OBSERVED|awp_03_player_victory"
	}
	else_if = {
		limit = { has_country_flag = aemusa_ms_country_fe_war_player_defeat }
		log = "AEMUSA-MS|OBSERVED|awp_03_player_defeat"
	}
	else_if = {
		limit = { has_country_flag = aemusa_ms_country_fe_war_status_quo }
		log = "AEMUSA-MS|OBSERVED|awp_03_status_quo"
	}
	else = { log = "AEMUSA-MS|WAIT|awp_03_no_war_result" }
	log = "AEMUSA-MS|AWP-03|audit end; writes=none"
}
