# 仅战斗摧毁回调能写此事实；目标不存在、撤退或脚本清理都不算击败。
aemusa_ms_anchor_record_battle = {
	if = {
		limit = {
			aemusa_ms_exclusive_active = yes
			check_variable = {
				which = aemusa_ms_country_anchor_stage
				value = 30
			}
			has_country_flag = aemusa_ms_country_anchor_projection_spawned

		}

		change_variable = {
			which = aemusa_ms_country_projections_defeated
			value = 1
		}

		log = "AEMUSA-MS|AWP-06|projection_defeated_in_combat"
	}
}
