building_artifact_coordination_center = {
	can_build = no
	base_buildtime = 480
	icon = "building_artifact_sanctum"
	category = government
	building_sets = { government }
	planet_limit = 1
	prerequisites = { "tech_artifact_covenant_cogovernance" }
	potential = { exists = owner owner = { owner_main_species = { is_species_class = "ARTIFACT_USER" } } }
	resources = {
		category = planet_buildings
		cost = {
			minerals = 1200 unity = 600
		}
		upkeep = {
			energy = 8 consumer_goods = 4
		}
		upkeep = {
			trigger = { artifact_garrison_coordination_active = yes }
			energy = 4
			consumer_goods = 1
		}
		# 高级统筹设施优先承接一份联防，不另加军队名额。
		upkeep = {
			trigger = {
				owner = { has_technology = "tech_artifact_civilization_joint_defense" }
				artifact_joint_defense_coordination_active = yes
			}
			energy = 4
			alloys = 1
		}
	}
	planet_modifier = {
		job_artifact_black_gate_warden_add = 300
		planet_stability_add = 4
		planet_amenities_add = 10
	}
	triggered_planet_modifier = {
		potential = { artifact_garrison_coordination_active = yes }
		planet_defense_armies_add = 2
	}
	triggered_planet_modifier = {
		potential = {
			owner = { has_technology = "tech_artifact_civilization_joint_defense" }
			artifact_joint_defense_coordination_active = yes
		}
		army_defense_damage_mult = 0.25
		army_defense_health_mult = 0.25
	}
	# 两座高级设施并存时由统筹中心供给唯一一份共治收益。
	triggered_planet_modifier = {
		potential = {
			has_active_building = "building_artifact_coordination_center"
			planet = { artifact_civil_supply_original_owner = yes artifact_civil_supply_safe = yes }
			owner = { has_technology = "tech_artifact_covenant_cogovernance" }
			has_modifier = acs_civic_industry
			NOT = { has_modifier = acs_civic_adjusting }
		}
		planet_jobs_alloys_produces_mult = 0.10
		planet_jobs_minerals_upkeep_mult = 0.05
	}
	# 两座高级设施并存时由统筹中心供给唯一一份共治收益。
	triggered_planet_modifier = {
		potential = {
			has_active_building = "building_artifact_coordination_center"
			planet = { artifact_civil_supply_original_owner = yes artifact_civil_supply_safe = yes }
			owner = { has_technology = "tech_artifact_covenant_cogovernance" }
			has_modifier = acs_civic_research
			NOT = { has_modifier = acs_civic_adjusting }
		}
		planet_jobs_physics_research_produces_mult = 0.05
		planet_jobs_society_research_produces_mult = 0.05
		planet_jobs_engineering_research_produces_mult = 0.05
		planet_jobs_consumer_goods_upkeep_mult = 0.05
	}
	# 两座高级设施并存时由统筹中心供给唯一一份共治收益。
	triggered_planet_modifier = {
		potential = {
			has_active_building = "building_artifact_coordination_center"
