            has_country_flag = ac_yutong_recruited
            has_country_flag = ac_rabi_recruited

        ac_refresh = yes

                days = 28
            }
            ac_seth_monthly = yes
            ac_yutong_monthly = yes
            ac_rabi_apply_protection = yes
        }
    }
}

# 本机原版：this为实际参战舰队，from为对手。先转国家事件再使用国家根作用域。
fleet_event = {
    id = artifact_companions.10
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        exists = owner
        owner = {
            ac_player = yes
            OR = {
                has_country_flag = ac_yutong_active
                has_country_flag = ac_rabi_active
            }
        }
        exists = from
        from = { exists = owner }
    }
    immediate = {
        save_event_target_as = ac_combat_force
        from = { save_event_target_as = ac_combat_threat }
        owner = { country_event = { id = artifact_companions.11 } }
    }
}
country_event = {
    id = artifact_companions.11
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        ac_player = yes
        exists = event_target:ac_combat_force
        exists = event_target:ac_combat_threat
        event_target:ac_combat_threat = { owner = { NOT = { is_same_value = root } } }
    }
    immediate = {
        ac_refresh = yes
        if = {
            limit = {
                has_country_flag = ac_yutong_active
                has_country_flag = ac_yutong_stable
                ac_yutong_in_position = yes
                event_target:ac_combat_force = { is_same_value = event_target:ac_yutong_force@root }
                NOR = {
                    has_country_flag = ac_yutong_recovery
                    has_country_flag = ac_yutong_contact
                    has_country_flag = ac_yutong_refused
                }
            }
            set_timed_country_flag = {
                flag = ac_yutong_contact
                days = 30
            }
            country_event = { id = artifact_companions.211 }
            log = "COMPANIONS|yutong_actual_combat"
        }
        if = {
            limit = {
                has_country_flag = ac_rabi_active
                ac_rabi_on_duty = yes
                event_target:ac_combat_force = { is_same_value = event_target:ac_rabi_force@root }
                NOT = { has_country_flag = ac_rabi_threat_registered }
            }
            ac_rabi_clear_threat = yes
            event_target:ac_combat_threat = {
                set_fleet_flag = ac_rabi_threat@root
                save_global_event_target_as = ac_rabi_threat@root
            }
            set_timed_country_flag = {
                flag = ac_rabi_threat_registered
                days = 90
            }
            ac_rabi_apply_protection = yes
            if = {
                limit = { ac_rabi_counter_available = yes }
                country_event = { id = artifact_companions.320 }
            }
            log = "COMPANIONS|rabi_actual_threat"
        }
    }
}
