# 首次选定路线后展示固定决定，可继续对应危机的实际回应。
country_event = {
	id = aemusa_ms.1015
	title = aemusa_ms.1015.title
	desc = {
		trigger = {
			check_variable = { which = aemusa_ms_country_vanilla_crisis_route_index value = 1 }
		}
		text = aemusa_ms.1015.prethoryn
	}
	desc = {
		trigger = {
			check_variable = { which = aemusa_ms_country_vanilla_crisis_route_index value = 2 }
		}
		text = aemusa_ms.1015.unbidden
	}
	desc = {
		trigger = {
			check_variable = { which = aemusa_ms_country_vanilla_crisis_route_index value = 3 }
		}
		text = aemusa_ms.1015.contingency
	}
	desc = {
		trigger = {
			check_variable = { which = aemusa_ms_country_vanilla_crisis_route_index value = 4 }
		}
		text = aemusa_ms.1015.cetana
	}
	is_triggered_only = yes
	picture = GFX_evt_aemusa_white_night_pavilion
	trigger = {
		aemusa_ms_country_can_enter_crisis_preparation = yes
		has_country_flag = aemusa_ms_country_history_vanilla_crisis_locked
	}
	option = {
		name = aemusa_ms.1015.close
		default_hide_option = yes
	}
	option = {
		name = aemusa_ms.1015.continue
		hidden_effect = {
			if = {
				limit = { aemusa_ms_country_has_crisis_route = yes }
				country_event = { id = aemusa_ms.1100 }
			}
		}
	}
}
