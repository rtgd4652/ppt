namespace = artifact_foundation
# 月度仅扫描带本包管理标记的自有殖民地，不排队补跑旧月份。
country_event = {
	id = artifact_foundation.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		any_owned_colony = {
			has_carrier_flag = arf_r1_managed
		}
	}
	immediate = {
		every_owned_colony = {
			limit = {
				has_carrier_flag = arf_r1_managed
			}
			# 月度枚举返回殖民地，界面事件使用行星载体；统一载体以保持动态目标与变量索引。
			carrier = {
				artifact_foundation_monthly_planet = yes
			}
		}
	}
}

# 4.5.2回调为殖民地载体；控制权变化不等于拥有者改变。
carrier_event = {
	id = artifact_foundation.2
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_carrier_flag = arf_r1_managed
	}
	immediate = {
		if = {
			limit = {
				artifact_foundation_original_owner = no
			}
			artifact_foundation_abort = yes
		}
	}
}
