# 序章 AED-PRO-00：从一局公平的牌开始，而不是从神谕或终局能力开始。
country_event = {
	id = aemusa_ms.200
	title = aemusa_ms.200.title
	desc = aemusa_ms.200.desc
	is_triggered_only = yes
	picture = GFX_evt_aemusa_white_night_pavilion
	trigger = {
		aemusa_ms_country_can_enter_prologue = yes
	}
	immediate = {
		# 写入前再次核对章节与前置，强制打开事件也不能补写未来事实。
		if = {
			limit = {
				aemusa_ms_country_can_enter_prologue = yes
			}
			aemusa_ms_begin_story_cluster = {
				STARTED_FLAG = aemusa_ms_country_cluster_aed_pro_00_started
				IN_PROGRESS_FLAG = aemusa_ms_country_cluster_aed_pro_00_in_progress
				COMPLETED_FLAG = aemusa_ms_country_cluster_aed_pro_00_completed
				SETTLED_FLAG = aemusa_ms_country_cluster_aed_pro_00_settled
				SETTLEMENT_FLAG = aemusa_ms_country_settlement_aed_pro_00_done
			}
		}
	}
	option = {
		name = aemusa_ms.200.continue
		# 内部状态仅参与结算，不向玩家展示变量与事件簇标记。
		hidden_effect = {
			# 整组选项副作用只执行一次；旧窗口只能关闭，不能追加历史或重新排队。
			if = {
				limit = {
					aemusa_ms_country_can_enter_prologue = yes
					aemusa_ms_country_can_complete_story_cluster = {
						STARTED_FLAG = aemusa_ms_country_cluster_aed_pro_00_started
						IN_PROGRESS_FLAG = aemusa_ms_country_cluster_aed_pro_00_in_progress
						COMPLETED_FLAG = aemusa_ms_country_cluster_aed_pro_00_completed
						SETTLED_FLAG = aemusa_ms_country_cluster_aed_pro_00_settled
						SETTLEMENT_FLAG = aemusa_ms_country_settlement_aed_pro_00_done
					}
				}
				aemusa_ms_finalize_story_cluster_once = {
					STARTED_FLAG = aemusa_ms_country_cluster_aed_pro_00_started
					IN_PROGRESS_FLAG = aemusa_ms_country_cluster_aed_pro_00_in_progress
					COMPLETED_FLAG = aemusa_ms_country_cluster_aed_pro_00_completed
					SETTLED_FLAG = aemusa_ms_country_cluster_aed_pro_00_settled
					SETTLEMENT_FLAG = aemusa_ms_country_settlement_aed_pro_00_done

				country_event = {
					id = aemusa_ms.210
				}
			}

	}
