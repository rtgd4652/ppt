namespace = aemusa_fp

# 开发入口：开启舰种、刷新自动设计并启动状态局势。
country_event = {
	id = aemusa_fp.1
	title = aemusa_fp.1.name
	desc = aemusa_fp.1.desc
	picture = GFX_evt_mercenary_fleet
	is_triggered_only = yes

	immediate = {
		set_country_flag = aemusa_fp_enabled
		if = {
			limit = { NOT = { is_variable_set = aemusa_fp_lifecycle_state } }
			aemusa_fp_set_lifecycle_state_effect = { STATE = 10 }
			set_variable = { which = aemusa_fp_destroyed_serial value = 0 }
			set_variable = { which = aemusa_fp_processed_serial value = 0 }
		}
		refresh_auto_generated_ship_designs = yes
		aemusa_fp_start_status_situation_effect = yes
	}

	option = {
		name = aemusa_fp.1.a
		country_event = { id = aemusa_fp.2 days = 1 }
	}
}

# 等待自动设计刷新后建立首个唯一实例。
country_event = {
	id = aemusa_fp.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = aemusa_fp_enabled
		NOT = { any_owned_ship = { is_ship_size = aemusa_fp_flagship } }
	}

	immediate = {
		aemusa_fp_set_lifecycle_state_effect = { STATE = 20 }
		aemusa_fp_spawn_flagship_effect = yes
		aemusa_fp_rebuild_ledger_effect = yes
		if = {
			limit = { any_owned_ship = { is_ship_size = aemusa_fp_flagship } }
			set_country_flag = aemusa_fp_first_build_completed
			aemusa_fp_set_lifecycle_state_effect = { STATE = 30 }
		}
		else = {
			aemusa_fp_set_lifecycle_state_effect = { STATE = 10 }
		}
	}
}

# 人工审计入口：按照实际舰船数量重建账本并显示结果。
country_event = {
	id = aemusa_fp.10
	title = aemusa_fp.10.name
	desc = {
		trigger = {
			count_owned_ship = {
				count = 1
				limit = { is_ship_size = aemusa_fp_flagship }
			}
		}
		text = aemusa_fp.10.desc_ok
	}
	desc = {
		trigger = {
			count_owned_ship = {
				count > 1
				limit = { is_ship_size = aemusa_fp_flagship }
			}
		}
		text = aemusa_fp.10.desc_duplicate
	}
	desc = { text = aemusa_fp.10.desc_missing }
	picture = GFX_evt_mercenary_fleet
	is_triggered_only = yes

	immediate = { aemusa_fp_rebuild_ledger_effect = yes }

	option = {
		name = aemusa_fp.10.recover
		trigger = {
			NOT = { any_owned_ship = { is_ship_size = aemusa_fp_flagship } }
		}

		# 开发期补偿入口：继续既有生命周期，不新增摧毁序号，也不生成第二艘合法旗舰。
		# 审计命令允许修复旧测试存档中遗失的启用标记，因此按钮不再以该标记作为显示前提。
		set_country_flag = aemusa_fp_enabled
		set_country_flag = aemusa_fp_flagship_reconstructing
		aemusa_fp_set_lifecycle_state_effect = { STATE = 50 }
		# 先接管标记舰队中唯一的无标记原型舰；确实没有候选时才补发新舰。
		aemusa_fp_adopt_single_unmarked_candidate_effect = yes
		if = {
			limit = { NOT = { any_owned_ship = { has_ship_flag = aemusa_fp_unique_flagship } } }
			aemusa_fp_spawn_flagship_effect = yes
		}
		aemusa_fp_rebuild_ledger_effect = yes
		if = {
			limit = {
				count_owned_ship = {
					count = 1
					limit = { has_ship_flag = aemusa_fp_unique_flagship }
				}
			}
			set_variable = { which = aemusa_fp_processed_serial value = aemusa_fp_destroyed_serial }
			set_country_flag = aemusa_fp_first_build_completed
			aemusa_fp_set_lifecycle_state_effect = { STATE = 30 }
			remove_country_flag = aemusa_fp_flagship_reconstructing
		}
		else = {
			aemusa_fp_set_lifecycle_state_effect = { STATE = 40 }
		}
		country_event = { id = aemusa_fp.10 days = 1 }
	}
	option = { name = aemusa_fp.close }
}

# 旧版部署锁定救援：只解除被 set_disabled 隐藏的原舰，不生成新舰，避免重复实例。

# 故障注入菜单：只供隔离原型验证恢复、重复和部署状态。
country_event = {
	id = aemusa_fp.20
	title = aemusa_fp.20.name
	desc = aemusa_fp.20.desc
	picture = GFX_evt_mercenary_fleet
	is_triggered_only = yes

	# 打开故障菜单时先按专属舰种重建运行身份，避免改装或读档后 ship flag 丢失使测试按钮失效。
	immediate = { aemusa_fp_rebuild_ledger_effect = yes }

	option = {
		name = aemusa_fp.20.destroy
		trigger = { any_owned_ship = { is_ship_size = aemusa_fp_flagship } }
		# 故障注入前先重建可能丢失的运行标记，确保测试与真实战斗走同一条重构事务。
		aemusa_fp_adopt_single_unmarked_candidate_effect = yes
		# destroy_ship 不保证触发战斗摧毁 on_action；故障注入先显式登记事务，再删除测试舰。
		aemusa_fp_begin_reconstruction_effect = yes
		random_owned_ship = {
			limit = { has_ship_flag = aemusa_fp_unique_flagship }
			destroy_ship = this
		}
	}
	option = {
		name = aemusa_fp.20.silent_delete
		trigger = { any_owned_ship = { is_ship_size = aemusa_fp_flagship } }
		aemusa_fp_adopt_single_unmarked_candidate_effect = yes
		random_owned_ship = {
			limit = { has_ship_flag = aemusa_fp_unique_flagship }
			fleet = { delete_fleet = this }
		}
		country_event = { id = aemusa_fp.110 days = 1 }
	}
	option = {
		name = aemusa_fp.20.duplicate
		aemusa_fp_spawn_flagship_effect = yes
		country_event = { id = aemusa_fp.10 days = 1 }
	}
	option = {
		name = aemusa_fp.20.deploy
		trigger = { NOT = { has_country_flag = aemusa_fp_flagship_deployed } }
		set_country_flag = aemusa_fp_flagship_deployed
		every_owned_ship = {
			limit = { is_ship_size = aemusa_fp_flagship }
			# 先清除部署前已经排入队列的移动、跃迁与其他舰队行动。
			fleet = {
				clear_orders = yes
				clear_fleet_actions = this
			}
			# 静态修正仅负责降低部署期间的意外位移，不再单独承担命令锁定。
			remove_modifier = aemusa_fp_deployment_lock
			add_modifier = { modifier = aemusa_fp_deployment_lock days = -1 }
		}
		# 监视器只在部署期间运行，并且每个国家最多保留一条事件链。
		if = {
			limit = { NOT = { has_country_flag = aemusa_fp_deployment_watchdog_running } }
			set_country_flag = aemusa_fp_deployment_watchdog_running
			country_event = { id = aemusa_fp.21 days = 1 }
		}
	}
	option = {
		name = aemusa_fp.20.recall
		trigger = { has_country_flag = aemusa_fp_flagship_deployed }
		remove_country_flag = aemusa_fp_flagship_deployed
		every_owned_ship = {
			limit = { is_ship_size = aemusa_fp_flagship }
			remove_modifier = aemusa_fp_deployment_lock
		}
	}
	option = {
		name = aemusa_fp.close
	}
}

# 部署命令监视器：每天只检查本国唯一旗舰并撤销玩家新下达的移动或跃迁命令。
# 原版没有“下达舰队命令”on_action，因此这里用单国单舰的轻量事件链替代全银河轮询。
country_event = {
	id = aemusa_fp.21
	hide_window = yes
	is_triggered_only = yes

	trigger = { has_country_flag = aemusa_fp_enabled }

	immediate = {
		if = {
			limit = { has_country_flag = aemusa_fp_flagship_deployed }
			every_owned_ship = {
				limit = { is_ship_size = aemusa_fp_flagship }
				fleet = {
					clear_orders = yes
					clear_fleet_actions = this
				}
			}
			country_event = { id = aemusa_fp.21 days = 1 }
		}
		else = {
			remove_country_flag = aemusa_fp_deployment_watchdog_running
		}
	}
}

# 摧毁监听：只接受带原型唯一标记的目标，并进入延迟重构状态。
country_event = {
	id = aemusa_fp.100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = aemusa_fp_enabled
		fromfrom = {
			OR = {
				has_ship_flag = aemusa_fp_unique_flagship
				is_ship_size = aemusa_fp_flagship
			}
		}
	}

	immediate = {
		aemusa_fp_begin_reconstruction_effect = yes
	}
}

# 延迟重构：若期间已经出现有效实例则只修复账本，不再生成第二艘。
country_event = {
	id = aemusa_fp.101
	hide_window = yes
	is_triggered_only = yes

	trigger = { has_country_flag = aemusa_fp_enabled }

	immediate = {
		# 以序号差判断是否存在未提交的真正摧毁事务，避免重复回调生成第二艘。
		set_variable = { which = aemusa_fp_pending_serial_delta value = aemusa_fp_destroyed_serial }
		subtract_variable = { which = aemusa_fp_pending_serial_delta value = aemusa_fp_processed_serial }
		if = {
			limit = {
				check_variable = { which = aemusa_fp_pending_serial_delta value > 0 }
				NOT = { any_owned_ship = { is_ship_size = aemusa_fp_flagship } }
			}
			aemusa_fp_set_lifecycle_state_effect = { STATE = 50 }
			aemusa_fp_spawn_flagship_effect = yes
		}
		aemusa_fp_rebuild_ledger_effect = yes
		if = {
			limit = {
				check_variable = { which = aemusa_fp_pending_serial_delta value > 0 }
				count_owned_ship = {
					count = 1
					limit = { has_ship_flag = aemusa_fp_unique_flagship }
				}
			}
			set_variable = { which = aemusa_fp_processed_serial value = aemusa_fp_destroyed_serial }
			aemusa_fp_set_lifecycle_state_effect = { STATE = 30 }
			remove_country_flag = aemusa_fp_flagship_reconstructing
		}
		clear_variable = aemusa_fp_pending_serial_delta
	}
}
